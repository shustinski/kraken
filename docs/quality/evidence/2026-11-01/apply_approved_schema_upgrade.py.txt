from pathlib import Path
import hashlib,json,os
from sqlalchemy import create_engine,text
from kraken_server.configuration import ServerConfig
from live_server_context import RUN, connect

config=ServerConfig.load(Path(os.environ['LOCALAPPDATA'])/'Kraken/LocalServer/server.toml')
engine=create_engine(config.database_url,hide_parameters=True)
receipt={'migration':'20261025_0010 -> 20261029_0011','authorization':'explicit user approval','database':'kraken_local'}
with engine.begin() as connection:
    connection.execute(text("SET LOCAL lock_timeout = '5s'"))
    connection.execute(text('LOCK TABLE background_jobs IN ACCESS EXCLUSIVE MODE'))
    versions=list(connection.execute(text('SELECT version_num FROM alembic_version FOR UPDATE')).scalars())
    assert versions==['20261025_0010'], versions
    count=connection.execute(text('SELECT count(*) FROM background_jobs')).scalar_one()
    assert count <= 10000, 'Review snapshot size before migration'
    before=[dict(r) for r in connection.execute(text('SELECT * FROM background_jobs ORDER BY job_id')).mappings()]
    connection.execute(text('ALTER TABLE background_jobs ADD COLUMN execution_state JSON'))
    updated=connection.execute(text("UPDATE alembic_version SET version_num='20261029_0011' WHERE version_num='20261025_0010'"))
    assert updated.rowcount==1
    after=[dict(r) for r in connection.execute(text('SELECT * FROM background_jobs ORDER BY job_id')).mappings()]
    for row in after:
        assert row.pop('execution_state') is None
    assert before==after, 'Existing job rows changed; rollback'
    receipt.update(previous_revision=versions[0],job_count=count,existing_job_rows_unchanged=True,new_column_nullable=True)
with engine.connect() as connection:
    receipt['committed_revision']=connection.execute(text('SELECT version_num FROM alembic_version')).scalar_one()
    column=connection.execute(text("SELECT data_type,is_nullable FROM information_schema.columns WHERE table_schema='public' AND table_name='background_jobs' AND column_name='execution_state' ")).one()
    assert tuple(column)==('json','YES')
remote=connect()
pid='7c622c09-73fa-552f-9a52-e80b64aeb3ef'
jobs=remote.plugin_jobs(pid)
receipt.update(jobs_http_status=200,project_jobs_count=len(jobs),project_revision=remote.get_project(pid).revision,status='passed')
(RUN/'schema-upgrade-working.json').write_text(json.dumps(receipt,indent=2),encoding='utf-8',newline='\n')
print(json.dumps(receipt))
engine.dispose()
